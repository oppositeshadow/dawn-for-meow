## 1. 技术栈全景与系统拓扑

```text
┌────────────────────────────────────────────────────────────────────────┐
│                        前端展示层 (Vue 3 客户端)                        │
│  - 框架：Vue 3 (Composition API / <script setup>) + Vite + TypeScript  │
│  - 样式：Tailwind CSS (暖白色/米黄 Cozy 卡片风格)                      │

│  - 状态：Pinia (按模块分仓存储各子系统响应式状态)                      │

                                    │ HTTP JSON 交互 (RESTful)
                                    │ (每 15 秒静默 POST /api/game/save 快照)
                                    │ (大盘/黑市/远征异步交互按需触发)
                                    ▼
┌────────────────────────────────────────────────────────────────────────┐
│                        后端业务层 (FastAPI 服务)                       │
│  - 框架：FastAPI (Python 3.10+) + Uvicorn (轻量、高性能异步 Web 框架)   │
│  - ORM / 数据库驱动：SQLAlchemy 2.0 (Async) + aiomysql / PyMySQL       │
│  - 校验：Pydantic v2 (严格类型校验与 Schema 约束)                      │
│  - 核心引擎：                                                          │
│    ├─ 确定性离线收益结算器 (Deterministic Offline Progress Engine)     │
│    ├─ 漂移布朗运动股市推演器 (Drift-Brownian Market Engine)            │
│    └─ 本地狂欢节语料渲染器 (Mad-libs Template Engine)                  │
└───────────────────┬────────────────────────────────┬───────────────────┘
                    │ 读写数据库 (耗时 < 1ms)        │ 异步调用 (结构化 JSON)
                    ▼                                ▼
┌──────────────────────────────────────┐  ┌──────────────────────────────┐
│        数据持久化层 (MySQL 8.0)      │  │     第三方 LLM API 网关       │
│  - 7 张紧凑核心业务表                │  │  - DeepSeek / OpenAI API     │
│  - 单表单行覆盖行星状态，纯数字池化  │  │  - 严格三道安全防火墙兜底    │
│  - JSON 字段存储格盘与持仓数组       │  │  - 仅用于新行星/特化卡/做局  │
└──────────────────────────────────────┘  └──────────────────────────────┘
```

## 3. 数据库设计全集 (MySQL DDL 与字段详解)

```sql
-- ========================================================
-- 1. 星球避难所核心状态表 (colony_state)
-- 单行星仅占一行数据，纯数字池化，纳秒级读写
-- ========================================================
CREATE TABLE colony_state (
    planet_id INT PRIMARY KEY COMMENT '星球标识: 0-翠绿母星, 1-熔岩星, 2-冰卫星, 3-小行星带矿区',
    last_save_time BIGINT NOT NULL COMMENT '最后存盘时间戳 (Unix Timestamp, 秒级，用于离线结算)',
    
    -- 基础存量物理资源 (当前存量 / 仓储上限)
    catnip DOUBLE DEFAULT 100.0 COMMENT '野生猫薄荷存量',
    catnip_max DOUBLE DEFAULT 500.0 COMMENT '猫薄荷仓储上限',
    scrap DOUBLE DEFAULT 0.0 COMMENT '机械废铁存量',
    scrap_max DOUBLE DEFAULT 300.0 COMMENT '废铁仓储上限',
    chips DOUBLE DEFAULT 0.0 COMMENT '工业逻辑芯片存量',
    chips_max DOUBLE DEFAULT 100.0 COMMENT '芯片仓储上限',
    alloys DOUBLE DEFAULT 0.0 COMMENT '航空钛合金存量',
    alloys_max DOUBLE DEFAULT 50.0 COMMENT '合金仓储上限',
    
    -- 流量平衡与隐匿对抗
    power_net DOUBLE DEFAULT 0.0 COMMENT '避难所净电力平衡 (kW)',
    suspicion DOUBLE DEFAULT 0.0 COMMENT '天网警戒度 (0.0 ~ 100.0)',
    tech_tier INT DEFAULT 1 COMMENT '基地科研等级 (每升一级，后续科技基准耗时按比例缩减)',
    
    -- 人口池化分桶 (纯数字统筹，杜绝个体对象与寿命遍历)
    total_cats INT DEFAULT 5 COMMENT '基地总猫口',
    birth_progress DOUBLE DEFAULT 0.0 COMMENT '自然繁育累加进度 (0.0 ~ 1.0)',
    job_farmers INT DEFAULT 3 COMMENT '猫草水培农夫工位数',
    job_scavengers INT DEFAULT 2 COMMENT '废土拾荒猫工位数',
    job_geeks INT DEFAULT 0 COMMENT '极客科研猫工位数',
    job_power_runners INT DEFAULT 0 COMMENT '踩轮发电猫工位数',
    job_idle INT DEFAULT 0 COMMENT '空闲待命猫口',
    
    -- 自动化换班协议配置 (JSON 存储)
    labor_automation_policy JSON DEFAULT NULL COMMENT '迟滞换班配置: {"enabled":true,"upper_limit":0.8,"lower_limit":0.2,"shift_count":2}',
    
    -- 设施建设等级 (工位上限与建筑状态)
    housing_boxes INT DEFAULT 5 COMMENT '恒温纸箱窝数量 (决定承载力 K)',
    cat_condo INT DEFAULT 0 COMMENT '多层豪华猫爬架数量',
    turing_terminals INT DEFAULT 1 COMMENT '图灵终端等级 (决定极客猫工位上限)',
    farm_plots INT DEFAULT 3 COMMENT '水培农田等级 (决定农夫工位上限)',
    scavenge_stations INT DEFAULT 2 COMMENT '解体操作台等级 (决定拾荒猫工位上限)',
    acoustic_layer INT DEFAULT 0 COMMENT '瓦楞纸隔音降噪层等级',
    launch_silo_stage INT DEFAULT 0 COMMENT '发射井总装阶段 (0-未动工, 1-竖坑清理, 2-导轨铸造, 3-推进剂加注, 4-点火就绪)'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='基地状态核心表';

-- ========================================================
-- 2. 科技研发记录表 (tech_records)
-- 维护 DAG 有向无环图节点研发状态
-- ========================================================
CREATE TABLE tech_records (
    id INT AUTO_INCREMENT PRIMARY KEY,
    planet_id INT NOT NULL COMMENT '所属星球 ID',
    tech_id VARCHAR(64) NOT NULL COMMENT '科技唯一标识 (如: tech_stealth_01, tech_laser)',
    tech_name VARCHAR(64) NOT NULL COMMENT '科技名称',
    status ENUM('LOCKED', 'RESEARCHING', 'UNLOCKED') DEFAULT 'LOCKED' COMMENT '研发状态',
    current_progress DOUBLE DEFAULT 0.0 COMMENT '当前已累积算力点数',
    target_cost DOUBLE NOT NULL COMMENT '解锁所需总算力点数',
    is_agent_generated BOOLEAN DEFAULT FALSE COMMENT '是否由 LLM 动态推演生成',
    buff_payload JSON DEFAULT NULL COMMENT '机制或数值加成效果载荷',
    UNIQUE KEY uk_planet_tech (planet_id, tech_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='科技树状态记录表';

-- ========================================================
-- 3. 水培基因实验室表 (garden_state)
-- 慢周期特色小游戏：网格状态与母本图鉴全收集
-- ========================================================
CREATE TABLE garden_state (
    planet_id INT PRIMARY KEY COMMENT '所属星球 ID',
    current_medium VARCHAR(32) DEFAULT 'STERILE' COMMENT '当前培养液: STERILE(无菌), RADIATION(放射催化), ZERO_G(零重力保鲜)',
    mechanical_arm_enabled BOOLEAN DEFAULT FALSE COMMENT '智能猫爪收割臂是否开启托管',
    auto_protect_unknown BOOLEAN DEFAULT TRUE COMMENT '是否自动保护未知突变种幼苗',
    grid_data JSON NOT NULL COMMENT '水培网格二维数组 [{"x":0,"y":0,"seed_id":"moss","stage":"mature","age":20}]',
    unlocked_seed_ids JSON NOT NULL COMMENT '已解锁母本基因图鉴列表 ["catnip", "moss", "gold_tear"]'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='水培基因实验室状态表';

-- ========================================================
-- 4. 智械匿名深网与股市大盘表 (darknet_state)
-- 快周期特色小游戏：Bitburner 概率大盘、做多/做空、黑市与暴露度
-- ========================================================
CREATE TABLE darknet_state (
    id INT PRIMARY KEY DEFAULT 1 COMMENT '单例全局表',
    byte_credits DOUBLE DEFAULT 200.0 COMMENT '暗网账户算力币余额',
    exposure DOUBLE DEFAULT 0.0 COMMENT '节点异常暴露度 (0.0 ~ 100.0)',
    burner_id VARCHAR(64) DEFAULT 'robot_4a932cz' COMMENT '当前匿名代号 (支持随机生成或玩家自定义)',
    has_4s_data BOOLEAN DEFAULT FALSE COMMENT '是否已购买 4S 深度数据眼透视插件',
    
    -- 股市行情快照 JSON: 记录价格、一阶隐藏胜率、二阶趋势锚点、动量
    -- [{"stock_id":"FORGE","name":"熔岩重工","price":45.2,"forecast":0.65,"target_forecast":0.4,"momentum":0.0,"volatility":0.035}]
    stocks_data JSON NOT NULL COMMENT '股市大盘实时状态快照',
    
    -- 借券做空持仓 JSON: 记录倒计时合约与借券利息
    -- [{"contract_id":"c1","stock_id":"FORGE","shares":100,"borrow_price":50.0,"expires_in":480,"leverage":2}]
    short_positions JSON NOT NULL COMMENT '借券做空合约持仓列表',
    
    -- 量化条件单配置 JSON: 自动高抛低吸
    -- [{"order_id":"o1","stock_id":"FORGE","type":"BUY_LONG","trigger_prob":0.7,"take_profit":0.2,"stop_loss":0.1}]
    limit_orders JSON NOT NULL COMMENT '量化策略条件单配置',
    
    -- 黑市货架状态 JSON: 当前波动价格与词条
    -- [{"item_id":"scrap","base_price":2.0,"current_price":3.2,"tag":"HOT_BUY"}]
    black_market_items JSON NOT NULL COMMENT '黑市实时货架状态'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='智械深网金融与股市大盘表';

-- ========================================================
-- 5. 载具机库与异步远征状态表 (military_state)
-- 载具编制、急救舱队列与异步远征
-- ========================================================
CREATE TABLE military_state (
    planet_id INT PRIMARY KEY COMMENT '所属星球 ID',
    
    -- 完好载具数量与战略军备
    light_cars INT DEFAULT 0 COMMENT '轻装猫车完好数',
    heavy_mechs INT DEFAULT 0 COMMENT '破拆机甲完好数',
    laser_turrets INT DEFAULT 0 COMMENT '固定激光防空炮台数',
    cruise_missiles INT DEFAULT 0 COMMENT '已总装地对地战略巡航导弹数',
    
    -- 损坏维修与急救舱队列 (JSON 数组存储)
    -- [{"unit_type":"light_car","cats_injured":2,"repair_remaining_seconds":180}]
    hospital_queue JSON NOT NULL COMMENT '猫猫急救舱休养排队',
    
    -- 异步远征任务队列 (JSON 数组存储)
    -- [{"expedition_id":"e1","target":"SUPERMARKET","cars":1,"crew":2,"remaining_seconds":300}]
    active_expeditions JSON NOT NULL COMMENT '当前进行中的远征任务',
    
    -- 三级安防预案配置 (JSON)
    -- {"p1_use_decoy":true,"p2_auto_intercept":true,"p3_auto_godark":true}
    security_policy JSON NOT NULL COMMENT '三级异步安防预案配置'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='载具机库与远征防务表';

-- ========================================================
-- 6. 敌对势力与宿敌 BOSS 状态表 (boss_state)
-- 欧米伽天网动态演化、物流后勤与前哨袭扰
-- ========================================================
CREATE TABLE boss_state (
    id INT PRIMARY KEY DEFAULT 1 COMMENT '单例全局表',
    threat_level INT DEFAULT 1 COMMENT '欧米伽威胁等级 (1~10)',
    expansion_rate DOUBLE DEFAULT 1.0 COMMENT '无人工厂扩张速率',
    fleet_strength DOUBLE DEFAULT 500.0 COMMENT '欧米伽核心舰队总战力',
    suspicion_toward_player DOUBLE DEFAULT 0.0 COMMENT '对玩家的锁定戒备度',
    
    -- 动态 RTS 物流与产线监控字段
    convoy_timer INT DEFAULT 1200 COMMENT '矿石补给车队下一次出发倒计时 (秒)',
    factory_frozen_seconds INT DEFAULT 0 COMMENT '兵工厂因被伏击缺料停工剩余秒数',
```

## 6. 后端核心业务推演引擎原型 (Python)

### 6.1 离线收益确定性计算引擎 (`offline_engine.py`)
```python
def process_offline_progress(colony_data: dict, elapsed_seconds: float) -> dict:
    """
    根据离线时间与基地配置，确定性结算资源积累、绝食罢工与科技繁育推进
    """
    total_cats = colony_data["total_cats"]
    farmers = colony_data["job_farmers"]
    scavengers = colony_data["job_scavengers"]
    geeks = colony_data["job_geeks"]
    
    # 1. 猫薄荷产出与消耗计算
    consumption_rate = total_cats * 0.2
    production_rate = farmers * 1.5
    net_catnip_rate = production_rate - consumption_rate
    
    current_catnip = colony_data["catnip"]
    catnip_max = colony_data["catnip_max"]
    
    # 2. 判定断粮时间点与绝食罢工
    starvation_occurred = False
    effective_work_seconds = elapsed_seconds
    
    if net_catnip_rate < 0:
        seconds_until_empty = current_catnip / abs(net_catnip_rate)
        if seconds_until_empty < elapsed_seconds:
            starvation_occurred = True
            effective_work_seconds = seconds_until_empty  # 断粮前正常工作
            final_catnip = 0.0
        else:
            final_catnip = max(0.0, current_catnip + net_catnip_rate * elapsed_seconds)
    else:
        final_catnip = min(catnip_max, current_catnip + net_catnip_rate * elapsed_seconds)
        
    # 3. 基础建材结算 (废铁)
    scrap_gain = scavengers * 0.5 * effective_work_seconds
    final_scrap = min(colony_data["scrap_max"], colony_data["scrap"] + scrap_gain)
    
    # 4. 科研算力结算 (绝食后算力冻结归 0)
    tech_tier_buff = 1.0 + (colony_data.get("tech_tier", 1) - 1) * 0.2
    research_gained = geeks * 1.0 * tech_tier_buff * effective_work_seconds
    
    # 5. 繁育进度结算 (S型曲线)
    K = colony_data["housing_boxes"] * 2
    r = 0.05 if not starvation_occurred else 0.0
    delta_birth = r * total_cats * (1.0 - min(1.0, total_cats / max(1, K))) * effective_work_seconds
    
    total_progress = colony_data["birth_progress"] + delta_birth
    new_born_cats = int(total_progress)
    final_birth_progress = total_progress - new_born_cats
    
    return {
        "final_catnip": round(final_catnip, 2),
        "final_scrap": round(final_scrap, 2),
        "research_gained": round(research_gained, 2),
        "new_born_cats": new_born_cats,
        "final_birth_progress": round(final_birth_progress, 4),
        "starvation_occurred": starvation_occurred,
        "elapsed_seconds": elapsed_seconds
    }
```

### 6.2 股市漂移布朗运动与均值回归算法 (`stock_engine.py`)
```python
import random

def advance_stock_tick(stock: dict) -> dict:
    """
    单只股票宏观周期步进算法 (结合胜率、引力与动量)
    """
    price = stock["price"]
    base_price = stock["base_price"]
    forecast = stock["forecast"]
    momentum = stock["momentum"]
    volatility = stock["volatility"]
    
    # 1. 均值回归引力 (Pull: 偏离过大时强力拉回基准线)
    pull = (base_price - price) * 0.025
    
    # 2. 一阶胜率掷骰子 (布朗随机扰动)
    is_up = random.random() < forecast
    noise_magnitude = random.gauss(0, volatility * price)
    noise = abs(noise_magnitude) if is_up else -abs(noise_magnitude)
    
    # 3. 动量冲击与合成新价格
    new_price = price + momentum + pull + noise
    stock["price"] = max(1.0, round(new_price, 2))
    
    # 4. 动量随时间衰减
    stock["momentum"] *= 0.90
    
    # 5. 二阶趋势锚点钟摆振荡 (胜率反转)
    target_forecast = stock["target_forecast"]
    stock["forecast"] += (target_forecast - forecast) * 0.03
    
    # 极值反转保护
    if stock["forecast"] > 0.85:
        stock["target_forecast"] = 0.25
    elif stock["forecast"] < 0.15:
        stock["target_forecast"] = 0.75
        
    return stock
```

```python
async def generate_special_tech_with_failsafe(planet_biome: str, unlocked_tech_ids: list[str]) -> dict:
    """
```

```python
import random
import re

CAT_NAMES = ["橘猫·铁柱", "三花·泡泡", "奶牛·薛定谔", "狸花·二踢脚", "暹罗·居里"]

    0 API 消耗、纳秒级本地狂欢节槽位替换

        "{building}": random.choice(BUILDINGS),
```
